document.addEventListener("DOMContentLoaded", () => {
    // 1. Integración Obligatoria con Autenticación (Seguridad de Rutas)
    const sesionActiva = JSON.parse(localStorage.getItem("sesion_sgg"));
    if (!sesionActiva) {
        window.location.href = "index.html";
        return;
    }

    // Elementos del DOM
    const userWelcome = document.getElementById("userWelcome");
    const btnLogout = document.getElementById("btnLogout");
    const gastoForm = document.getElementById("gastoForm");
    const gastoIdInput = document.getElementById("gastoId");
    const montoInput = document.getElementById("monto");
    const fechaInput = document.getElementById("fecha");
    const categoriaSelect = document.getElementById("categoria");
    const descripcionInput = document.getElementById("descripcion");
    const tablaGastosBody = document.getElementById("tablaGastosBody");
    const totalGastadoDisplay = document.getElementById("totalGastado");
    const btnCancelar = document.getElementById("btnCancelar");
    const modalEliminar = document.getElementById("modalEliminar");
    const btnConfirmarEliminar = document.getElementById("btnConfirmarEliminar");
    const btnCancelarEliminar = document.getElementById("btnCancelarEliminar");

    let gastoIdAEliminar = null;

    userWelcome.textContent = `Usuario: ${sesionActiva.email}`;

    // 2. Control de Cerrar Sesión
    btnLogout.addEventListener("click", () => {
        localStorage.removeItem("sesion_sgg");
        window.location.href = "index.html";
    });

    // 3. Lectura y Filtrado por Usuario Activo y Baja Lógica (RF-06 y RF-09)
    const obtenerGastosLocalStorage = () => {
        return JSON.parse(localStorage.getItem("gastos_sgg")) || [];
    };

    const guardarGastosLocalStorage = (gastos) => {
        localStorage.setItem("gastos_sgg", JSON.stringify(gastos));
    };

    const renderizarInterfaz = () => {
        const todosLosGastos = obtenerGastosLocalStorage();
        
        // Filtrar vinculando con el usuario logueado (Foreign Key) y estado activo
        const gastosUsuario = todosLosGastos.filter(
            (gasto) => gasto.usuario_id === sesionActiva.email && gasto.estado_activo === true
        );

        // Renderizar Tabla
        tablaGastosBody.innerHTML = "";
        gastosUsuario.forEach((gasto) => {
            const tr = document.createElement("tr");
            tr.innerHTML = `
                <td>${gasto.fecha}</td>
                <td>${gasto.categoria}</td>
                <td>${gasto.descripcion}</td>
                <td>$${parseFloat(gasto.monto).toFixed(2)}</td>
                <td>
                    <button type="button" onclick="prepararEdicion(${gasto.id})">Editar</button>
                    <button type="button" onclick="solicitarEliminacion(${gasto.id})">Eliminar</button>
                </td>
            `;
            tablaGastosBody.appendChild(tr);
        });

        // Recálculo del Resumen del Dashboard
        const total = gastosUsuario.reduce((acc, curr) => acc + parseFloat(curr.monto), 0);
        totalGastadoDisplay.textContent = `$${total.toFixed(2)}`;
    };

    // 4. Carga y Edición de Gastos (RF-05 y RF-07)
    gastoForm.addEventListener("submit", (e) => {
        e.preventDefault();

        const monto = parseFloat(montoInput.value);
        if (isNaN(monto) || monto <= 0) {
            alert("El monto ingresado debe ser mayor a cero.");
            return;
        }

        const gastos = obtenerGastosLocalStorage();
        const idActual = gastoIdInput.value;

        if (idActual) {
            // Actualización (Update)
            const index = gastos.findIndex((g) => g.id == idActual);
            if (index !== -1) {
                gastos[index].monto = monto;
                gastos[index].fecha = fechaInput.value;
                gastos[index].categoria = categoriaSelect.value;
                gastos[index].descripcion = descripcionInput.value;
            }
        } else {
            // Carga Nueva (Create)
            const nuevoGasto = {
                id: Date.now(),
                usuario_id: sesionActiva.email, // Vinculación Clave Foránea
                monto: monto,
                fecha: fechaInput.value,
                categoria: categoriaSelect.value,
                descripcion: descripcionInput.value,
                estado_activo: true // Baja Lógica Inicial
            };
            gastos.push(nuevoGasto);
        }

        guardarGastosLocalStorage(gastos);
        limpiarFormulario();
        renderizarInterfaz();
    });

    window.prepararEdicion = (id) => {
        const gastos = obtenerGastosLocalStorage();
        const gasto = gastos.find((g) => g.id === id);
        if (gasto) {
            gastoIdInput.value = gasto.id;
            montoInput.value = gasto.monto;
            fechaInput.value = gasto.fecha;
            categoriaSelect.value = gasto.categoria;
            descripcionInput.value = gasto.descripcion;

            document.getElementById("formTitle").textContent = "Editar Gasto";
            btnCancelar.style.display = "inline-block";
        }
    };

    btnCancelar.addEventListener("click", limpiarFormulario);

    function limpiarFormulario() {
        gastoIdInput.value = "";
        gastoForm.reset();
        document.getElementById("formTitle").textContent = "Registrar Nuevo Gasto";
        btnCancelar.style.display = "none";
    }

    // 5. Baja Lógica / Eliminación Segura (RF-08)
    window.solicitarEliminacion = (id) => {
        gastoIdAEliminar = id;
        modalEliminar.showModal();
    };

    btnConfirmarEliminar.addEventListener("click", () => {
        if (gastoIdAEliminar) {
            const gastos = obtenerGastosLocalStorage();
            const index = gastos.findIndex((g) => g.id === gastoIdAEliminar);
            
            if (index !== -1) {
                // Cambio de estado_activo a false (sin destructores como splice)
                gastos[index].estado_activo = false;
                guardarGastosLocalStorage(gastos);
            }

            gastoIdAEliminar = null;
            modalEliminar.close();
            renderizarInterfaz();
        }
    });

    btnCancelarEliminar.addEventListener("click", () => {
        gastoIdAEliminar = null;
        modalEliminar.close();
    });

    // Inicialización al cargar la vista
    renderizarInterfaz();
});
